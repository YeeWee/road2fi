    ```tsx
    import { getPostBySlug, getAllPostSlugs } from "@/lib/posts"
    import Link from "next/link"
    import type { Metadata } from "next"

    export async function generateStaticParams() {
      const slugs = await getAllPostSlugs()
      return slugs.map((slug) => ({ slug }))
    }

    export async function generateMetadata({
      params,
    }: {
      params: Promise<{ slug: string }>
    }): Promise<Metadata> {
      const { slug } = await params
      const post = await getPostBySlug(slug)
      if (!post) {
        return { title: "Post Not Found — Road to Fi" }
      }
      return {
        title: `${post.title} — Road to Fi`,
        description: post.excerpt,
      }
    }

    export default async function PostPage({
      params,
    }: {
      params: Promise<{ slug: string }>
    }) {
      const { slug } = await params
      const post = await getPostBySlug(slug)

      if (!post) {
        return (
          <section className="py-20">
            <div className="mx-auto max-w-5xl px-4 sm:px-6 lg:px-8 text-center">
              <h1 className="text-2xl font-bold text-gray-900">Post not found</h1>
              <p className="mt-4 text-gray-600">
                The post you're looking for doesn't exist.
              </p>
              <Link href="/blog" className="mt-6 inline-block text-blue-600 hover:text-blue-700">
                ← Back to blog
              </Link>
            </div>
          </section>
        )
      }

      return (
        <article className="py-16 sm:py-20">
          <div className="mx-auto max-w-5xl px-4 sm:px-6 lg:px-8">
            {/* Header */}
            <header className="mb-10 text-center">
              <time className="text-sm text-gray-500">{post.date}</time>
              <h1 className="mt-2 text-3xl sm:text-4xl font-bold tracking-tight text-gray-900">
                {post.title}
              </h1>
            </header>

            {/* Thumbnail */}
            {post.thumbnail && (
              <div className="mb-10">
                <img
                  src={post.thumbnail}
                  alt={post.title}
                  className="w-full rounded-xl object-cover"
                  style={{ maxHeight: "400px" }}
                />
              </div>
            )}

            {/* Markdown content — rendered as plain text inside prose for v1 */}
            <div className="prose prose-gray max-w-none">
              {post.content}
            </div>

            {/* Back link */}
            <div className="mt-12 pt-8 border-t border-gray-200">
              <Link href="/blog" className="text-sm text-gray-600 hover:text-gray-900">
                ← Back to blog
              </Link>
            </div>
          </div>
        </article>
      )
    }
    ```

    Key decisions:
    - Next.js 16 uses `params` as a Promise — must `await params` to get `{ slug }`
    - `generateStaticParams()` pre-renders all known slugs at build time
    - `generateMetadata()` sets page title and description from post frontmatter
    - Markdown content rendered as `{post.content}` inside `prose` div — renders as styled text (headings, paragraphs appear as formatted blocks via prose CSS)
    - Not using `dangerouslySetInnerHTML` — content is plain text rendered inside prose-styled div. For v1 this is acceptable; Markdown formatting (## headings, **bold**) will display as literal text but with prose typography applied.
    - Actually, `post.content` from gray-matter is the raw Markdown string. Rendering it directly as `{post.content}` in JSX will show it as a single text node — NOT parsed Markdown. We need to either:
      a) Use `dangerouslySetInnerHTML` with a Markdown-to-HTML converter, OR
      b) Split content by newlines and render each line with appropriate element

    Since we have NO Markdown parser installed (gray-matter only extracts frontmatter), and we cannot add dependencies, use option (b): simple line-by-line rendering.

    ```tsx
    function MarkdownRenderer({ content }: { content: string }) {
      const lines = content.split("\n")
      const elements: React.ReactNode[] = []
      let inList = false
      let listItems: string[] = []

      const flushList = () => {
        if (listItems.length > 0) {
          elements.push(
            <ul key={`list-${elements.length}`} className="list-disc pl-6 space-y-1">
              {listItems.map((item, i) => <li key={i}>{item}</li>)}
            </ul>
          )
          listItems = []
        }
        inList = false
      }

      for (const line of lines) {
        const trimmed = line.trim()

        // Empty line → skip (paragraph break handled by spacing)
        if (trimmed === "") {
          flushList()
          continue
        }

        // Heading 2: ## Title
        if (trimmed.startsWith("## ")) {
          flushList()
          elements.push(
            <h2 key={elements.length} className="text-2xl font-bold text-gray-900 mt-8 mb-4">
              {trimmed.replace("## ", "")}
            </h2>
          )
          continue
        }

        // Heading 3: ### Title
        if (trimmed.startsWith("### ")) {
          flushList()
          elements.push(
            <h3 key={elements.length} className="text-xl font-semibold text-gray-900 mt-6 mb-3">
              {trimmed.replace("### ", "")}
            </h3>
          )
          continue
        }

        // List item: - Item or * Item
        if (trimmed.startsWith("- ") || trimmed.startsWith("* ")) {
          inList = true
          listItems.push(trimmed.substring(2))
          continue
        }

        // Flush list if we were in one
        flushList()

        // Default → paragraph
        elements.push(
          <p key={elements.length} className="text-gray-700 leading-relaxed">
            {trimmed}
          </p>
        )
      }

      flushList()
      return <>{elements}</>
    }
    ```

    Then use it: `<MarkdownRenderer content={post.content} />`

    This is a pragmatic v1 approach — handles ## headings, ### headings, - list items, and paragraphs. No bold/italic/code block parsing (those would need a proper Markdown parser). For the content we control (generated blog posts), this is sufficient.
