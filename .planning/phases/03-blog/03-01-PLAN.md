    Props: Accept a single `post` prop of type `PostMeta` from `@/lib/posts`.

    Render structure:
    ```tsx
    import Link from "next/link"
    import type { PostMeta } from "@/lib/posts"

    interface PostCardProps {
      post: PostMeta
    }

    export default function PostCard({ post }: PostCardProps) {
      return (
        <Link href={`/blog/${post.slug}`} className="group block rounded-xl border border-gray-200 bg-white overflow-hidden shadow-sm transition-shadow hover:shadow-md">
          {/* Thumbnail image */}
          {post.thumbnail ? (
            <img
              src={post.thumbnail}
              alt={post.title}
              className="w-full h-48 object-cover"
            />
          ) : (
            <div className="w-full h-48 bg-gray-100 flex items-center justify-center">
              <span className="text-gray-400 text-sm">No image</span>
            </div>
          )}

          {/* Content */}
          <div className="p-6">
            <time className="text-xs text-gray-500">{post.date}</time>
            <h3 className="mt-2 text-lg font-semibold text-gray-900 group-hover:text-gray-700">
              {post.title}
            </h3>
            <p className="mt-2 text-sm text-gray-600 line-clamp-3">
              {post.excerpt}
            </p>
          </div>
        </Link>
      )
    }
    ```

    Design decisions:
    - The entire card is a Link (better UX — click anywhere to read)
    - `group` class on Link enables hover color change on title
    - `line-clamp-3` truncates excerpt to 3 lines (roughly 150 chars visually)
    - Placeholder gray box when no thumbnail (thumbnail can be empty string from frontmatter)
    - Uses same card styling pattern as Philosophy.tsx: `rounded-xl border border-gray-200 bg-white shadow-sm`
    - `overflow-hidden` ensures image doesn't bleed past rounded corners
  </action>

    This is a Server Component (default in App Router). No "use client" directive needed.

    ```tsx
    import { getAllPosts } from "@/lib/posts"
    import PostCard from "@/components/blog/PostCard"

    export default function BlogPage() {
      const posts = getAllPosts()

      return (
        <section className="py-16 sm:py-20">
          <div className="mx-auto max-w-5xl px-4 sm:px-6 lg:px-8">
            <div className="mb-12 text-center">
              <h1 className="text-3xl sm:text-4xl font-bold tracking-tight text-gray-900">
                Blog
              </h1>
              <p className="mt-3 text-lg text-gray-600 max-w-2xl mx-auto">
                Insights and lessons from our journey toward Financial Independence.
              </p>
            </div>

            {posts.length === 0 ? (
              <p className="text-center text-gray-500">No posts yet. Check back soon!</p>
            ) : (
              <div className="grid grid-cols-1 gap-8 sm:grid-cols-2 lg:grid-cols-3">
                {posts.map((post) => (
                  <PostCard key={post.slug} post={post} />
                ))}
              </div>
            )}
          </div>
        </section>
      )
    }
    ```

    Design decisions:
    - `getAllPosts()` is called directly (sync — the function in lib/posts.ts uses fs.readFileSync internally despite the async signature; Next.js server components can await or call sync)
    - Actually, `getAllPosts()` returns `Promise<Post[]>`, so use `const posts = await getAllPosts()` and make the component async: `export default async function BlogPage()`
    - Grid: 1 column on mobile, 2 on sm, 3 on lg — responsive per BLOG-06
    - Empty state handled with friendly message
    - Section padding: `py-16 sm:py-20` consistent with Philosophy.tsx `py-20 sm:py-24`
    - Page title "Blog" with subtitle matching FIRE theme tone
    - No additional imports needed beyond what's shown
